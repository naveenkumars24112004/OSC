----------------------------------------------------
AIM
To design and implement a Convolutional Neural Network (CNN) using TensorFlow to classify handwritten digit images.

Algorithm
Step 1: Import the required libraries: TensorFlow, NumPy, Matplotlib, and Random.

Step 2: Load the MNIST handwritten digit dataset containing training and testing images.

Step 3: Display the dataset summary and sample handwritten digit images.

Step 4: Select a limited number of training and testing images for faster execution.

Step 5: Normalize the image pixel values by dividing them by 255.

Step 6: Reshape the images to include the channel dimension required by the CNN.

Step 7: Construct the CNN with:

Input layer

Convolutional layer with ReLU activation

Max Pooling layer

Flatten layer

Dense hidden layer with ReLU activation

Output layer with 10 neurons and Softmax activation

Step 8: Compile the CNN using the Adam optimizer, Sparse Categorical Cross-Entropy loss function, and accuracy metric.

Step 9: Train the CNN using the training images for 2 epochs with a batch size of 64.

Step 10: Evaluate the trained CNN using the testing images and display the test accuracy.

Step 11: Randomly select one image from the testing dataset.

Step 12: Predict the handwritten digit using the trained CNN.

Step 13: Display the actual digit, predicted digit, and selected handwritten digit image.

program:
# Handwritten Digit Classification using MNIST Dataset
import os
import sys
import warnings

os.environ["TF_CPP_MIN_LOG_LEVEL"] = "3"
os.environ["TF_ENABLE_ONEDNN_OPTS"] = "0"
warnings.filterwarnings("ignore")

stderr = sys.stderr
sys.stderr = open(os.devnull, "w")

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
import random

sys.stderr = stderr

# Load MNIST Dataset
(X_train, y_train), (X_test, y_test) = tf.keras.datasets.mnist.load_data()

# Display Dataset Summary
print("===== MNIST DATASET SUMMARY =====")
print(f"Total Training Images\t\t: {len(X_train)}")
print(f"Total Testing Images\t\t: {len(X_test)}")
print(f"Image Size\t\t\t: {X_train.shape[1]} x {X_train.shape[2]}")
print("Number of Classes\t\t: 10")
print("Classes\t\t\t: 0 to 9")

# Display Sample Images
plt.figure(figsize=(8, 4))
for i in range(10):
    plt.subplot(2, 5, i + 1)
    plt.imshow(X_train[i], cmap="gray")
    plt.title("Digit: " + str(y_train[i]))
    plt.axis("off")

plt.suptitle("Sample MNIST Handwritten Digits")
plt.tight_layout()
plt.show()

# Use Smaller Dataset for Faster Execution
X_train = X_train[:2000]
y_train = y_train[:2000]
X_test = X_test[:500]
y_test = y_test[:500]

print("\n===== DATA USED =====")
print("Training Images :", len(X_train))
print("Testing Images  :", len(X_test))

# Normalize Images
X_train = X_train / 255.0
X_test = X_test / 255.0

# Add Channel Dimension
X_train = X_train.reshape(-1, 28, 28, 1)
X_test = X_test.reshape(-1, 28, 28, 1)

# Create CNN Model
model = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(28, 28, 1)),
    tf.keras.layers.Conv2D(16, (3, 3), activation="relu"),
    tf.keras.layers.MaxPooling2D((2, 2)),
    tf.keras.layers.Flatten(),
    tf.keras.layers.Dense(32, activation="relu"),
    tf.keras.layers.Dense(10, activation="softmax")
])

# Compile Model
model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

# Train Model
print("\n===== MODEL TRAINING =====")
model.fit(X_train, y_train, epochs=2, batch_size=64)

# Evaluate Model
_, accuracy = model.evaluate(X_test, y_test, verbose=0)
print("\n===== MODEL PERFORMANCE =====")
print("Test Accuracy:", round(accuracy * 100, 2), "%")

# Select a Random Test Image
index = random.randint(0, len(X_test) - 1)

# Predict the Random Image
prediction = model.predict(X_test[index:index + 1], verbose=0)
predicted_digit = np.argmax(prediction[0])
actual_digit = y_test[index]

# Display Prediction
print("\n===== DIGIT PREDICTION =====")
print("Test Image No. :", index)
print("Actual Digit   :", actual_digit)
print("Predicted Digit:", predicted_digit)

# Display the Random Test Image
plt.imshow(X_test[index].reshape(28, 28), cmap="gray")
plt.title("Actual: " + str(actual_digit) + "\nPredicted: " + str(predicted_digit))
plt.axis("off")
plt.show()
