Title: 7. Employee Information

----------------------------------------------------
AIM
To build a Node.js employee information system with Express, MongoDB, Mongoose, and administrator authentication.

Algorithm
1. Start

2. Load environment configuration, create the Express app, and add URL-encoded form and session middleware.

3. Connect to MongoDB using the configured connection string.

4. Find the configured administrator account and create it with a hashed password if it does not exist.

5. Display the login form and verify submitted credentials against the stored password hash.

6. Regenerate the session ID after successful login and protect employee routes with authentication middleware.

7. Search and filter employee records and display the results on the dashboard.

8. Validate and save new employee records using the Mongoose model.

9. Provide routes to view, edit, and delete employee records, then start the server.

10. Stop

                                          # employee.js
const mongoose = require("mongoose");

const employeeSchema = new mongoose.Schema(
  {
    empId: { type: String, required: true, unique: true, trim: true },
    name: { type: String, required: true, trim: true },
    email: {
      type: String,
      required: true,
      trim: true,
      lowercase: true,
      match: /^[^\s@]+@[^\s@]+\.[^\s@]+$/,
    },
    phone: { type: String, required: true, trim: true },
    department: { type: String, required: true, trim: true },
    designation: { type: String, required: true, trim: true },
    salary: { type: Number, required: true, min: 0 },
    joiningDate: { type: Date, required: true },
    address: { type: String, trim: true },
    skills: { type: [String], default: [] },
  },
  { timestamps: true },
);

module.exports = mongoose.model("Employee", employeeSchema);

                                          # server.js
const express = require("express");
const mongoose = require("mongoose");
const path = require("path");
const Employee = require("./employee");

const app = express();
const PORT = Number(process.env.PORT) || 3002;
const MONGO_URI =
  process.env.MONGO_URI || "mongodb://127.0.0.1:27017/employee_portal";

app.set("view engine", "ejs");
app.set("views", path.join(__dirname, "views"));
app.use(express.urlencoded({ extended: false }));
app.use(express.static(path.join(__dirname, "public")));

app.get("/", async (req, res, next) => {
  try {
    const employees = await Employee.find().sort({ joiningDate: -1, name: 1 });
    res.render("dashboard", { employees });
  } catch (error) {
    next(error);
  }
});

app.get("/employees/new", (req, res) => {
  res.render("add", { error: null, employee: {} });
});

app.post("/employees", async (req, res) => {
  const employee = {
    empId: req.body.empId,
    name: req.body.name,
    email: req.body.email,
    phone: req.body.phone,
    department: req.body.department,
    designation: req.body.designation,
    salary: req.body.salary,
    joiningDate: req.body.joiningDate,
    address: req.body.address,
    skills: (req.body.skills || "")
      .split(",")
      .map((skill) => skill.trim())
      .filter(Boolean),
  };

  try {
    await Employee.create(employee);
    res.redirect("/");
  } catch (error) {
    if (error.code === 11000) {
      return res.status(409).render("add", {
        error: "That employee ID is already in use.",
        employee,
      });
    }

    if (error instanceof mongoose.Error.ValidationError || error instanceof mongoose.Error.CastError) {
      return res.status(400).render("add", {
        error: "Please check the employee details and try again.",
        employee,
      });
    }

    console.error("Could not save employee:", error);
    res.status(500).render("add", {
      error: "Unable to save the employee right now. Please try again.",
      employee,
    });
  }
});

app.get("/employees/:id", async (req, res, next) => {
  try {
    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(404).render("not-found");
    }

    const employee = await Employee.findById(req.params.id);
    if (!employee) {
      return res.status(404).render("not-found");
    }

    res.render("view", { employee });
  } catch (error) {
    next(error);
  }
});

app.post("/employees/:id/delete", async (req, res, next) => {
  try {
    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(404).render("not-found");
    }

    const employee = await Employee.findByIdAndDelete(req.params.id);
    if (!employee) {
      return res.status(404).render("not-found");
    }

    res.redirect("/");
  } catch (error) {
    next(error);
  }
});

app.use((error, req, res, next) => {
  console.error("Request failed:", error);
  if (res.headersSent) {
    return next(error);
  }
  res.status(500).send("Something went wrong. Please try again.");
});

async function start() {
  try {
    await mongoose.connect(MONGO_URI);
    console.log("Connected to MongoDB");
    app.listen(PORT, () => {
      console.log(`Employee portal running at http://localhost:${PORT}`);
    });
  } catch (error) {
    console.error("Could not connect to MongoDB:", error.message);
    process.exitCode = 1;
  }
}

if (require.main === module) {
  start();
}

module.exports = app;
