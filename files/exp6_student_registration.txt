Title: 6. Student Registration

----------------------------------------------------
AIM
To build a Node.js and Express student registration application that authenticates an administrator, validates student details, and displays the registered student information.

Algorithm
1. Start

2. Load Express and session middleware and serve the static files.

3. Display the administrator login form.

4. Check the submitted administrator credentials and create a login session if valid.

5. Require a valid session before serving the student registration form.

6. Read the submitted student details and check that required fields are not empty.

7. Validate the email address and ensure the mobile number contains exactly 10 digits.

8. Save the validated details and selected skills in the session.

9. Display the student details on the confirmation page and destroy the session on logout.

10. Stop

    fetch("/student-data")
     .then((response) => response.json())
     .then((student) => {
      const fields = [
        ["Student Name", student.name],
        ["Register Number", student.regno],
        ["Email", student.email],
        ["Mobile Number", student.mobile],
        ["Gender", student.gender],
        ["Date of Birth", student.dob],
        ["Department", student.department],
        ["Course", student.course],
        ["Year", student.year],
        ["Address", student.address],
        ["Skills", student.skills],
      ];
      const table = document.createElement("table");
      for (const [label, value] of fields) {
        const row = table.insertRow();
        const heading = document.createElement("th");
        const cell = row.insertCell();
        heading.textContent = label;
        cell.textContent = value ?? "";
        row.prepend(heading);
      }
      document.getElementById("studentDetails").replaceChildren(table);
     });
  </script>

                              # App.js

const express = require("express");
const session = require("express-session");
const path = require("path");
const fs = require("fs");

const app = express();
const PORT = process.env.PORT || 3001;
const viewsPath = path.join(__dirname, "views");
const publicPath = path.join(__dirname, "public");
const pagePath = fs.existsSync(viewsPath) ? viewsPath : __dirname;
const assetPath = fs.existsSync(publicPath) ? publicPath : __dirname;

// Middleware
app.use(express.urlencoded({ extended: true }));
app.use(express.static(assetPath));

app.use(
 session({
   secret: "student-registration-secret",
  resave: false,
  saveUninitialized: false,
 }),
);

// Login page
app.get("/", (req, res) => {
 res.sendFile(path.join(pagePath, "login.html"));
});

// Login
app.post("/login", (req, res) => {
 const { username, password } = req.body;

  if (username === "admin" && password === "1234") {
  req.session.loggedIn = true;
   res.redirect("/register");
 } else {
   res.send(`
       <h2>Invalid username or password</h2>
      <a href="/">Back to Login</a>
      `);
 }
});

// Registration page
app.get("/register", (req, res) => {
  if (!req.session.loggedIn) {
  return res.redirect("/");
 }

 res.sendFile(path.join(pagePath, "register.html"));
});

// Handle registration
app.post("/register", (req, res) => {
  if (!req.session.loggedIn) {
  return res.redirect("/");
 }

 const {
  name,
  regno,
  email,
  mobile,
  gender,
  dob,
  department,
  course,
  year,
  address,
  skills,
 } = req.body;

  // Validation
  if (
  !name ||
  !regno ||
  !email ||
  !mobile ||
  !gender ||
  !dob ||
  !department ||
  !course ||
  !year ||
  !address
 ) {
  return res.send(`
      <h2>All required fields must be filled.</h2>
      <a href="/register">Go Back</a>
      `);
 }

  // Email validation
 const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

  if (!emailPattern.test(email)) {
  return res.send(`
       <h2>Invalid email address.</h2>
      <a href="/register">Go Back</a>
      `);
 }

  // Mobile validation
 const mobilePattern = /^[0-9]{10}$/;

  if (!mobilePattern.test(mobile)) {
  return res.send(`
      <h2>Mobile number must contain exactly 10 digits.</h2>
      <a href="/register">Go Back</a>
      `);
 }

  // Store student details in session
 req.session.student = {
  name,
  regno,
  email,
  mobile,
  gender,
  dob,
  department,
  course,
  year,
  address,
  skills: skills || "None",
 };

 res.redirect("/student");
});

// Display student details
app.get("/student", (req, res) => {
  if (!req.session.loggedIn) {
  return res.redirect("/");
 }

  if (!req.session.student) {
  return res.redirect("/register");
 }

 const student = req.session.student;

 res.sendFile(path.join(pagePath, "student.html"));
});

// Student data API used by student.html
app.get("/student-data", (req, res) => {
  if (!req.session.loggedIn) {
  return res.status(401).json({
    error: "Unauthorized",
   });
 }

 res.json(req.session.student);
});

// Logout
app.get("/logout", (req, res) => {
 req.session.destroy(() => {
   res.redirect("/");
 });
});

app.listen(PORT, () => {
 console.log(`Server running at http://localhost:${PORT}`);
});
